export function statusClass(status: Status) {
  return {
    PASS: "status-pass",
    FAIL: "status-fail",
    REVIEW: "status-review",
    UNAVAILABLE: "status-unavailable"
  }[status];
}

export function riskClass(risk: Risk) {
  return {
    LOW: "risk-low",
    MEDIUM: "risk-medium",
    HIGH: "risk-high",
    CRITICAL: "risk-critical"
  }[risk];
}

export function formatPercent(value: number) {
  return `${Math.round(value * 100)}%`;
}

export async function ingestDocument(fileName: string) {
  return {
    fileName,
    algorithm: "SHA-3-512",
    hash: "9b7f4d8a1f342c18b7b6d2e4c9a8e21c",
    status: "sealed",
    timestamp: "08 Sep 2026, 10:42 IST"
  };
}

export async function evaluateBidder(bidderId: string) {
  const bidder = bidders.find((item) => item.id === bidderId);
  if (!bidder) throw new Error("Bidder not found");
  return bidder;
}

export async function verifyIdentifier(source: string, value: string) {
  return {
    source,
    value,
    status: source === "GST" ? "CONNECTED" : "UNAVAILABLE",
    result: source === "GST" ? "PASS" : "UNAVAILABLE",
    decisionImpact: source === "GST" ? "No change" : "REVIEW"
  };
}
